5/// Add the high lane and subtract the low lane.
6///
7/// * **Intrinsic:** [`_mm_addsub_pd`]
8/// * **Assembly:** `addsubpd xmm, xmm`
9#[must_use]
10#[inline(always)]
11#[cfg_attr(docsrs, doc(cfg(target_feature = "sse3")))]
12pub fn addsub_m128d(a: m128d, b: m128d) -> m128d {
13  m128d(unsafe { _mm_addsub_pd(a.0, b.0) })
14}
