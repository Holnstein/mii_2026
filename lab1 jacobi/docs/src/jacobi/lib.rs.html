37    #[test]
38    /// Проверка формулы (7): при выборе угла по (9)-(11)
39    fn rotation_check_seven_equation_from_lab1() {
40        let cases = [
41            (7.0, 19.0, 13.0),               // пример из методички, C > 0
42            (19.0, 7.0, 13.0),               // C < 0
43            (7.0, 19.0, -13.0),              // a_pq < 0
44            (100.0, -100.0, 50.0),           // C = 0
45        ];
46
47        for (a_pp, a_qq, a_pq) in cases {
48            let (cos, sin) = rotation_cos_sin(a_pp, a_qq, a_pq)
49                .expect("a_pq != 0, должен быть Some");
50
51            // формула (6): A_pq' = (cos^2 - sin^2)·a_pq + sin * cos * (a_pp - a_qq)
52            let new_a_pq = (cos * cos - sin * sin) * a_pq
53                + sin * cos * (a_pp - a_qq);
54
55            let eps = 1e-5;
56
57            assert!(
58                new_a_pq.abs() < eps,
59                "A_pq' = {new_a_pq}, ожидалось около 0 с точностью {eps} для \
60                 a_pp = {a_pp}, a_qq = {a_qq}, a_pq = {a_pq}"
61            );
62        }
63    }

65    #[test]
66    // проверка поворота матрицы A, будет падать на маленьких eps из-за сравнения матриц
67    // по методичке
68    fn rotate_a_example_lab1() {
69        let mut a = DMatrix::from_row_slice(5, 5, &[
70            1.0, 4.0, 7.0, 10.0, 13.0,
71            4.0, 7.0, 10.0, 13.0, 16.0,
72            7.0, 10.0, 13.0, 16.0, 19.0,
73            10.0, 13.0, 16.0, 19.0, 22.0,
74            13.0, 16.0, 19.0, 22.0, 25.0,
75        ]);
76
77        let (cos, sin) = rotation_cos_sin(a[(1, 1)], a[(3, 3)], a[(1, 3)]).unwrap();
78        let tg = sin / cos;
79
80        rotate_a(&mut a, 1, 3, tg, cos, sin);
81
82        assert_eq!(a[(1, 3)], 0.0, "A[1][3] должен быть 0");
83        assert_eq!(a[(3, 1)], 0.0, "A[4][2] должен быть 0");
84
85        let a_itog = DMatrix::from_row_slice(5, 5, &[
86            1.0, -2.02, 7.0, 10.58, 13.0,
87            -2.02, -1.32, -0.2, 0.0, 1.62,
88            7.0, -0.2, 13.0, 18.86, 19.0,
89            10.58, 0.0, 18.86, 27.32, 27.15,
90            13.0, 1.62, 19.0, 27.15, 25.0,
91        ]);
92
93        for r in 0..a.nrows() {
94            for c in 0..a.ncols() {
95                assert!(
96                    (a[(r, c)] - a[(c, r)]).abs() < 1e-5,
97                    "симметрия нарушена в [{r}][{c}]: {} vs {}",
98                    a[(r, c)], a[(c, r)]
99                );
100            }
101        }
102
103        let is_equal = relative_eq!(a, a_itog, epsilon = 1e-2);
104        assert!(is_equal, "a={a} a itog = {a_itog}");
105    }
